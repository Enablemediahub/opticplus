    ADD COLUMN `patient_type` VARCHAR(10) NULL AFTER `purpose`,

To backfill legacy classifications manually after adding the column, run:
  UPDATE `patient_records`
  SET `patient_type` = CASE
    WHEN LOWER(`comment`) LIKE '%new patient%' AND LOWER(`comment`) NOT LIKE '%old patient%' THEN 'new'
    WHEN LOWER(`comment`) LIKE '%old patient%' AND LOWER(`comment`) NOT LIKE '%new patient%' THEN 'old'
    ELSE `patient_type`
  END
  WHERE `patient_type` IS NULL
    AND `comment` IS NOT NULL
    AND (LOWER(`comment`) LIKE '%new patient%' OR LOWER(`comment`) LIKE '%old patient%');
